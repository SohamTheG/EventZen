Project EventZen {
  database_type: 'MySQL'
  Note: 'Event Management System - Distributed Microservices Schema'
}

// ==========================================
// 1. USER & ATTENDEE DATABASE (IAM)
// ==========================================
TableGroup "userattendeesdb" {
  users
  attendees
}

Table users {
  id bigint [pk, increment]
  email varchar(255) [not null]
  password varchar(255)
  name varchar(255)
  role varchar(255)
}

Table attendees {
  id bigint [pk, increment]
  user_id bigint
  event_id int [not null]
  status varchar(255)
}

Ref: attendees.user_id > users.id


// ==========================================
// 2. EVENTS & BOOKING DATABASE (Transactional)
// ==========================================
TableGroup "events_booking_db" {
  events
  bookings
}

Table events {
  id bigint [pk, increment]
  name varchar(255)
  description varchar(255)
  host_id bigint 
  is_public bit [not null]
  ticket_price double [not null, default: 0]
}

Table bookings {
  id bigint [pk, increment]
  event_date date
  status varchar(255) [note: "enum('APPROVED','CANCELLED','PENDING','REJECTED')"]
  venue_id bigint 
  event_id bigint [unique]
  total_amount_due double [not null, default: 0]
}

Ref: bookings.event_id - events.id


// ==========================================
// 3. VENUE & VENDOR DATABASE (Logistics)
// ==========================================
TableGroup "venue_service_db" {
  Venues
  Vendors
  VenueVendors
}

Table Venues {
  id bigint [pk, increment]
  name varchar(255) [not null]
  location varchar(255) [not null]
  capacity int
  price_per_day decimal(10,2)
  is_available boolean [default: true]
  createdAt datetime 
  updatedAt datetime 
}

Table Vendors {
  id bigint [pk, increment]
  name varchar(255) [not null]
  type varchar(255) [not null, note: "e.g., 'Catering', 'Decor'"]
  service_fee decimal(10,2) [default: 0.00]
  createdAt datetime
  updatedAt datetime
}

Table VenueVendors {
  VenueId bigint [pk]
  VendorId bigint [pk]
  createdAt datetime
  updatedAt datetime
}

Ref: VenueVendors.VenueId > Venues.id
Ref: VenueVendors.VendorId > Vendors.id


// ==========================================
// 4. CROSS-DATABASE SOFT LINKS
// (Application Layer Aggregation Only - No physical SQL constraints)
// ==========================================

// Links the Booking Event to the Identity User (Host)
Ref: events.host_id > users.id

// Links the Attendee record to the Booking Event
Ref: attendees.event_id > events.id

// Links the Booking record to the Physical Venue
Ref: bookings.venue_id > Venues.id
