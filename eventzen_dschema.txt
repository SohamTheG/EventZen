Table users {
  id int [pk, increment]
  email varchar [unique]
  password varchar
  name varchar
  role varchar
}

Table attendees {
  id int [pk, increment]
  user_id int [ref: - users.id]
  event_id int
  status varchar
  qr_code_base64 string
  quantity int
  ticket_identifier string [unique]
}

Table events {
  id int [pk, increment]
  name varchar
  description varchar
  host_id int [ref: - users.id]
  is_public bit
  ticket_price decimal
}

Table bookings {
  id int [pk, increment]
  event_date date
  status enum [APPROVED, CANCELLED, PENDING, REJECTED]
  venue_id int [ref: - venues.id]
  event_id int [unique, ref: - events.id]
  total_amount_due decimal
}

Table Venues {
  id int [pk, increment]
  name varchar
  location varchar
  capacity int
  price_per_day decimal
  is_available bit
  createdAt datetime
  updatedAt datetime
}

Table Vendors {
  id int [pk, increment]
  name varchar
  type varchar
  createdAt datetime
  updatedAt datetime
  service_fee decimal
}

Table VenueVendors {
  VenueId int [ref: - Venues.id]
  VendorId int [ref: - Vendors.id]
  createdAt datetime
  updatedAt datetime
  
  indexes {
    (VenueId, VendorId) [pk]
  }
}
